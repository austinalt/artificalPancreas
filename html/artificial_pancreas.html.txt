<script>
  const analyzeBtn = document.getElementById('analyzeBtn');
  const bloodSugarInput = document.getElementById('bloodSugarInput');
  const resultDiv = document.getElementById('result');

  // Function to add glow animation class briefly
  function pulseGlow() {
    resultDiv.classList.add('glow');
    setTimeout(() => resultDiv.classList.remove('glow'), 2500);
  }

  function analyzeBloodSugar(level) {
    let status = '';
    let instruction = '';

    // Using typical clinical ranges:
    // Low: under 70
    // Normal: 70 - 180
    // High: above 180

    if (level < 70) {
      status = 'Low Blood Sugar (Hypoglycemia)';
      instruction = `Your blood sugar is too low. <strong>Immediate action:</strong> Consume fast-acting carbohydrates such as glucose tablets, fruit juice, or regular soda. Monitor levels closely and consult your healthcare provider if symptoms persist.`;
    } else if (level <= 180) {
      status = 'Normal Blood Sugar';
      instruction = `Your blood sugar level is within a healthy range. Maintain your routine and keep monitoring regularly. Stay hydrated and follow your treatment plan.`;
    } else {
      status = 'High Blood Sugar (Hyperglycemia)';
      instruction = `Your blood sugar is elevated. <strong>Suggested steps:</strong> Check for ketones if possible, stay hydrated, and avoid high-carb foods temporarily. Contact your healthcare provider if you feel unwell or if this persists.`;
    }

    return {status, instruction};
  }

  analyzeBtn.addEventListener('click', () => {
    const val = bloodSugarInput.value.trim();
    const level = Number(val);

    if (!val || isNaN(level)) {
      resultDiv.innerHTML = `<div class="status normal">Please enter your blood sugar level as a number.</div>`;
      resultDiv.focus();
      return;
    }

    if (level < 20 || level > 600) {
      resultDiv.innerHTML = `<div class="status normal">Please enter a realistic blood sugar value between 20 and 600 mg/dL.</div>`;
      resultDiv.focus();
      return;
    }

    const analysis = analyzeBloodSugar(level);
    const statusClass = analysis.status.includes('Low') ? 'low' : (analysis.status.includes('High') ? 'high' : 'normal');

    resultDiv.innerHTML = `
      <div class="status ${statusClass}">${analysis.status}</div>
      <div class="instruction">${analysis.instruction}</div>
    `;

    pulseGlow();
    resultDiv.focus();
  });

  // Enter key triggers the analysis for better UX on mobile
  bloodSugarInput.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') {
      e.preventDefault();
      analyzeBtn.click();
    }
  });
</script>
